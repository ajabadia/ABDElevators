import { z } from 'zod';
import { ObjectId } from 'mongodb';

/**
 * TIPO BASE: EntityId
 * Siempre stringified ObjectId, nunca ObjectId nativo ni string arbitrario
 */
export const EntityIdSchema = z.string()
  .regex(/^[0-9a-fA-F]{24}$/, 'Must be valid MongoDB ObjectId')
  .brand<'EntityId'>();

export type EntityId = z.infer<typeof EntityIdSchema>;

/**
 * TIPO BASE: TenantScoped
 * Todas las entidades de negocio deben extender esto
 */
export const TenantScopedSchema = z.object({
  tenantId: EntityIdSchema,
  createdAt: z.date().default(() => new Date()),
  updatedAt: z.date().default(() => new Date()),
  deletedAt: z.date().optional(), // Soft delete universal
  deletedBy: EntityIdSchema.optional(),
});

export type TenantScoped = z.infer<typeof TenantScopedSchema>;

/**
 * TIPO BASE: VersionedEntity
 * Para documentos que evolucionan (prompts, workflows, etc.)

export const VersionedEntitySchema = z.object({

    version: z.number(),

    lastName: z.string(),
    avatarUrl: z.string().url().optional(),
    preferences: z.object({
      uxMode: z.enum(['simple', 'expert']).default('simple'),
      locale: z.string().default('es'),
      timezone: z.string().default('Europe/Madrid'),
      notifications: z.object({
        email: z.boolean().default(true),
        inApp: z.boolean().default(true),
        digestFrequency: z.enum(['immediate', 'daily', 'weekly']).default('immediate'),
      }),
    }),
  }),
  
  // MFA con historial de dispositivos
  mfa: z.object({
    enabled: z.boolean().default(false),
    method: z.enum(['totp', 'sms', 'email']).optional(),
    verifiedAt: z.date().optional(),
    backupCodesUsed: z.number().default(0),
    devices: z.array(z.object({
      deviceId: z.string(),
      name: z.string(),
      lastUsedAt: z.date(),
      ip: z.string(),
    })).default([]),
  }),
  
  // Auditoría de sesiones (para "Active Sessions" feature)
  sessions: z.array(z.object({
    sessionId: z.string(),
    createdAt: z.date(),
    expiresAt: z.date(),
    ip: z.string(),
    userAgent: z.string(),
    isCurrent: z.boolean().default(false),
  })).default([]),
  
  // Metadata del sistema
  ...TenantScopedSchema.shape,
  lastLoginAt: z.date().optional(),
  loginCount: z.number().default(0),
});
